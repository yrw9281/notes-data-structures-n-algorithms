CLASS Bag<T>
    CREATE an empty list named items

    METHOD add(item)
        Append item to items

    METHOD remove(item)
        Remove the first matching item from items, if present

    METHOD contains(item)
        Return whether item is present in items

    METHOD count(item)
        Set occurrences to 0
        FOR each currentItem in items
            IF currentItem equals item
                Increase occurrences by 1
        Return occurrences

    METHOD toString()
        Return items as a string

    METHOD main()
        Create an empty Bag of strings
        Add apple, banana, apple, orange, and apple
        Print the bag
        Test contains with apple and grape
        Test count for apple and grape
        Remove one apple
        Print the bag again
        Test contains for apple
        Test count for apple; the result should be 2
END CLASS
